    <script src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>

    </style>

<body>
    <div id="root"></div>

    <script type="text/babel">
        const { useState } = React;

        const slides = [
            { title: "Claude (AI-verktyg)", content: "Claude kan användas för text och skrivande, programmering, analys, problemlösning, idéer, kreativitet och dokumenthantering." },
            { title: "Varför jag använder Claude", content: "Jag har valt att använda Claude främst för att skapa kod snabbt och effektivt." },
            { title: "Fördelar", content: "Snabb, pedagogisk, bra för nybörjare och prototyper." },
            { title: "Nackdelar", content: "Kan ge fel kod, minskar inlärning om man bara kopierar, kräver kontroll och förståelse." },
            { title: "Viktigt att veta", content: "Claude ger ofta färdig kod utan förklaring. Du får koden och instruktioner för hur du använder den." },
            { title: "Priser", content: "Gratisversion med begränsade tokens. Pro: ca 15€/månad. Max: ca 90€/månad." },
            { title: "Prompt 1", content: "Skapa en Python-kod som genererar ett slumpmässigt nummer mellan 1–3000." },
            { title: "Resultat (Prompt 1)", content: "Funkade bra när koden kördes i Google Colab.", image: "bilder/slumpmässigt_tal.png" },
            { title: "Prompt 2", content: "Skapa ett Snake-spel i Python.", content: "Koden fungerade, men krävde pygame. När jag körde i Replit behövdes viss anpassning.", image: "bilder/snake_spel.png" }
        ];

        function Presentation() {
            const [index, setIndex] = useState(0);
            const slide = slides[index];

            return (
                <div className="min-h-screen flex flex-col items-center justify-center p-6">
                    <div className="w-full max-w-2xl bg-white shadow-xl rounded-2xl p-8 transition-all">
                        <h1 className="text-2xl font-bold mb-4">{slide.title}</h1>
                        <p className="text-base whitespace-pre-line mb-4">{slide.content}</p>
                        {slide.image && (
                            <img src={slide.image} alt={slide.title} className="rounded-xl shadow-md w-full object-contain max-h-96" />
                        )}
                    </div>
                    <div className="flex gap-4 mt-6">
                        <button 
                            className={`px-4 py-2 rounded bg-blue-500 text-white ${index === 0 ? 'opacity-50' : ''}`}
                            onClick={() => setIndex(Math.max(index - 1, 0))}
                            disabled={index === 0}>Bakåt</button>
                        <button 
                            className={`px-4 py-2 rounded bg-blue-500 text-white ${index === slides.length - 1 ? 'opacity-50' : ''}`}
                            onClick={() => setIndex(Math.min(index + 1, slides.length - 1))}
                            disabled={index === slides.length - 1}>Nästa</button>
                    </div>
                    <p className="mt-4 text-sm text-gray-500">Slide {index + 1} / {slides.length}</p>
                </div>
            );
        }

        const root = ReactDOM.createRoot(document.getElementById('root'));
        root.render(<Presentation />);
    </script>
